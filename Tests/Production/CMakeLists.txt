cmake_minimum_required(VERSION 3.19)
project(OrbiterProductionBootstrap LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "The WGL bootstrap must execute on native Windows")
endif()
add_executable(orbiter-wgl-probe WglProbe.cpp)
target_compile_features(orbiter-wgl-probe PRIVATE cxx_std_17)
target_compile_definitions(orbiter-wgl-probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(orbiter-wgl-probe PRIVATE opengl32 gdi32 user32 psapi)
if(MSVC)
    target_compile_options(orbiter-wgl-probe PRIVATE /W4 /EHsc)
endif()
