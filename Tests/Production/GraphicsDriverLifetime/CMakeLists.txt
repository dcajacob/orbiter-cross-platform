cmake_minimum_required(VERSION 3.24)
project(GraphicsDriverLifetime LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "This diagnostic uses Linux EGL and /proc/self/maps")
endif()
find_package(OpenGL REQUIRED COMPONENTS OpenGL EGL)
add_library(driver_fixture MODULE Fixture.cpp)
target_link_libraries(driver_fixture PRIVATE OpenGL::EGL OpenGL::OpenGL)
add_executable(driver_lifetime Loader.cpp)
target_link_libraries(driver_lifetime PRIVATE ${CMAKE_DL_LIBS})
foreach(target driver_fixture driver_lifetime)
    target_compile_features(${target} PRIVATE cxx_std_17)
    target_compile_options(${target} PRIVATE -g -Wall -Wextra -Werror -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
        target_compile_options(${target} PRIVATE -fno-gnu-unique)
    endif()
endforeach()
enable_testing()
foreach(mode no-mipmap mipmap)
    add_test(NAME driver-${mode} COMMAND driver_lifetime $<TARGET_FILE:driver_fixture> ${mode})
    set_tests_properties(driver-${mode} PROPERTIES
        ENVIRONMENT "LIBGL_ALWAYS_SOFTWARE=1;GALLIUM_DRIVER=llvmpipe;ASAN_OPTIONS=detect_leaks=1:fast_unwind_on_malloc=0"
        TIMEOUT 30)
endforeach()
