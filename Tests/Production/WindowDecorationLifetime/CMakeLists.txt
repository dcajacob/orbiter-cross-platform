cmake_minimum_required(VERSION 3.24)
project(OrbiterWindowDecorationLifetime LANGUAGES C CXX)
include(FetchContent)
set(ORBITER_GLFW_ARCHIVE "" CACHE FILEPATH "Pinned GLFW archive from the renderer dependency cache")
if(NOT EXISTS "${ORBITER_GLFW_ARCHIVE}")
    message(FATAL_ERROR "Supply the pinned renderer GLFW archive")
endif()
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -fsanitize=address,undefined -fno-omit-frame-pointer")
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -fsanitize=address,undefined -fno-omit-frame-pointer")
set(GLFW_BUILD_WAYLAND ON CACHE BOOL "" FORCE)
set(GLFW_BUILD_X11 OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
set(GLFW_LIBRARY_TYPE STATIC CACHE STRING "" FORCE)
FetchContent_Declare(glfw URL "${ORBITER_GLFW_ARCHIVE}"
    URL_HASH SHA256=5234f4f29473e9a06bc7847d8371858dd135d38466eeeaa652fdc9f8f9ff0c20
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(glfw)
add_executable(window-decoration-lifetime main.cpp)
target_compile_features(window-decoration-lifetime PRIVATE cxx_std_17)
target_link_libraries(window-decoration-lifetime PRIVATE glfw)
option(ORBITER_ICON_OWNER_PROBE "Build read-only GTK icon ownership observations" OFF)
if(ORBITER_ICON_OWNER_PROBE)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(ICON_GTK REQUIRED gtk+-3.0)
    add_library(window-icon-info-probe MODULE IconInfoProbe.c)
    target_include_directories(window-icon-info-probe PRIVATE ${ICON_GTK_INCLUDE_DIRS})
    target_compile_options(window-icon-info-probe PRIVATE ${ICON_GTK_CFLAGS_OTHER})
    target_link_libraries(window-icon-info-probe PRIVATE dl)
    add_executable(window-icon-owner IconOwner.cpp)
    target_compile_features(window-icon-owner PRIVATE cxx_std_17)
    target_link_libraries(window-icon-owner PRIVATE glfw)
endif()
include(CTest)
foreach(mode prefer disable)
    add_test(NAME wayland_decoration_${mode} COMMAND window-decoration-lifetime ${mode})
    set_tests_properties(wayland_decoration_${mode} PROPERTIES ENVIRONMENT
        "LIBGL_ALWAYS_SOFTWARE=1;GALLIUM_DRIVER=llvmpipe;LP_NUM_THREADS=2;ASAN_OPTIONS=detect_leaks=1:fast_unwind_on_malloc=0;LSAN_OPTIONS=;UBSAN_OPTIONS=halt_on_error=1:print_stacktrace=1")
endforeach()
