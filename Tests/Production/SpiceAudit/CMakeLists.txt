cmake_minimum_required(VERSION 3.19)
project(OrbiterSpiceDataAudit LANGUAGES CXX)
get_filename_component(ORBITER_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The production data audit requires x86-64")
endif()
if(MSVC)
    set(TOOLKIT win_x64)
    set(TOOLKIT_LIBRARY cspice.lib)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|amd64|AMD64")
    set(TOOLKIT linux_x64)
    set(TOOLKIT_LIBRARY cspice.a)
else()
    message(FATAL_ERROR "Only the pinned MSVC Windows and native Linux CSPICE libraries are supported")
endif()
set(CSPICE "${ORBITER_ROOT}/Src/Celbody/SPICE/cspice/${TOOLKIT}")
add_executable(orbiter-spice-data-audit ../SpiceCoverage.cpp)
target_compile_features(orbiter-spice-data-audit PRIVATE cxx_std_17)
target_include_directories(orbiter-spice-data-audit PRIVATE "${CSPICE}/include")
target_link_libraries(orbiter-spice-data-audit PRIVATE "${CSPICE}/lib/${TOOLKIT_LIBRARY}")
if(NOT MSVC)
    target_link_libraries(orbiter-spice-data-audit PRIVATE m)
endif()
