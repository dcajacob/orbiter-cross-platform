cmake_minimum_required(VERSION 3.24)
project(OrbiterSdkExample LANGUAGES CXX)
find_package(OrbiterSDK CONFIG REQUIRED)
if(WIN32)
    add_library(Phase8SdkProbe MODULE Vessel.cpp)
    orbiter_sdk_vessel(Phase8SdkProbe)
else()
    add_library(ShuttlePB MODULE ShuttlePB.cpp)
    orbiter_sdk_vessel(ShuttlePB)
endif()

if(DEFINED ORBITER_REJECTION_FIXTURE AND NOT WIN32)
    add_library(SdkMismatch MODULE "${ORBITER_REJECTION_FIXTURE}")
    target_compile_definitions(SdkMismatch PRIVATE PHASE6_REJECTED_METADATA=0)
    orbiter_sdk_vessel(SdkMismatch)
endif()
