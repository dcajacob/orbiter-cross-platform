cmake_minimum_required(VERSION 3.19)
project(OrbiterProjectLauncher LANGUAGES CXX)

find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets)
if(NOT TARGET orbiter-launchpad-ui)
    add_subdirectory(../Launchpad launchpad EXCLUDE_FROM_ALL)
endif()
qt_add_executable(orbiter-project-launcher WIN32 MACOSX_BUNDLE main.cpp)
target_link_libraries(orbiter-project-launcher PRIVATE orbiter-launchpad-ui)
set_target_properties(orbiter-project-launcher PROPERTIES MACOSX_BUNDLE_BUNDLE_NAME "Orbiter Project Launcher")

include(GNUInstallDirs)
install(TARGETS orbiter-project-launcher BUNDLE DESTINATION . RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    install(FILES orbiter-project-launcher.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
    install(FILES orbiter-project-launcher.png DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/64x64/apps)
endif()

include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 6.4 REQUIRED COMPONENTS Test)
    qt_add_executable(project-launcher-test RuntimeLibrary.Test.cpp)
    set_target_properties(project-launcher-test PROPERTIES AUTOMOC ON)
    target_compile_features(project-launcher-test PRIVATE cxx_std_17)
    target_link_libraries(project-launcher-test PRIVATE orbiter-launchpad-ui Qt6::Test)
    add_test(NAME project-launcher COMMAND project-launcher-test)
    set_tests_properties(project-launcher PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
